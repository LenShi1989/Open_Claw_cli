//**Ollama 相關**//

irm https://ollama.com/install.ps1 | iex
ollama --version                          //確認 Ollama 版本

//**Openclaw Cli相關**//

//安裝 OpenClaw
iwr -useb https://openclaw.ai/install.ps1 | iex  

openclaw --version                //查看 OpenClaw 版本
openclaw --help                   //查看說明
openclaw config                   //設定參數
openclaw status                   //查看目前 OpenClaw 狀態
openclaw gateway install          //安裝 Gateway
openclaw gateway status           //查看 Gateway 狀態
openclaw gateway start            //啟動 Gateway
openclaw gateway stop             //停止 Gateway
openclaw gateway restart          //重啟 Gateway
openclaw update                   //更新 OpenClaw 
openclaw agents add coding        //建立sub-agent, name為coding
openclaw doctor                   //檢查／診斷問題
openclaw logs --follow            //檢查及時log
openclaw config validate          //驗證設定檔
openclaw node status              //確認 Node Host 狀態
openclaw node install             //安裝 Node Host
openclaw node start               //啟動 Node Host
openclaw node restart             //重啟 Node Host
openclaw nodes list               //檢查 Gateway 是否看到 Node
openclaw nodes status             //檢查 Gateway 是否看到 Node
openclaw devices list             //查看那些回應授權  
openclaw nodes pending            //查看 nodes 暫停的檔案
openclaw skills list --verbose    //查看skills清單
openclaw skills check             //確認skills
openclaw models list              //查看現在的AI模型
openclaw models status            //查看AI模型狀態

openclaw config get agents.defaults.workspace              //查看目前工作區
openclaw config set agents.defaults.workspace "路徑"       //設定工作路徑
openclaw setup --workspace "路徑"                          //初始化工作區
openclaw config validate                                   //驗證設定
openclaw models list --provider ollama                     //查看 Ollama 
openclaw models set ollama/qwen3.5:9b                      //設定主要模型
openclaw models list --provider ollama                     //確認模型
openclaw models fallbacks add ollama/gemma4                //將模型加入備用 


openclaw devices approve 104a56fb-0ab8-4542-8610-f94a26f96468  //允許授權
openclaw nodes approve 5f76ddbb-fda7-4950-b990-85ff7c54d4fa    //允許授權

//設定timeout
openclaw config set models.providers.ollama.timeoutSeconds 600

//**openclaw內部指令**//
/help       //顯示可用指令              
/status     //查看目前 Agent / Session 狀態 
/model      //查看或切換模型                
/models     //查看可用模型                  
/reset      //重置目前 Session              
/clear      //清除目前對話內容              
/new        //建立新的 Session              
/sessions   //查看 Session                  
/session    //查看目前 Session 資訊         
/compact    //壓縮目前 Context              
/context    //查看 Context 使用狀況         
/usage      //查看 Token / 使用量           
/think      //調整思考模式                  
/verbose    //開／關詳細輸出                
/debug      //開／關 Debug 資訊             
/tools      //查看可用 Tools                
/skills     //查看可用 Skills               
/config     //查看設定                      
/quit       //離開目前 Agent                
/exit       //離開 Agent
